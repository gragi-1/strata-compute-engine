cmake_minimum_required(VERSION 3.20)
project(strata_worker_cpp VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(STRATA_BUILD_WORKER "Build the gRPC and Docker worker (Linux dependencies required)" ON)
enable_testing()
add_executable(lease_tracker_test tests/lease_tracker_test.cpp)
target_include_directories(lease_tracker_test PRIVATE include)
add_test(NAME lease_tracker COMMAND lease_tracker_test)
add_executable(wave_solver ../workloads/wave_solver/main.cpp)
add_test(NAME wave_solver COMMAND wave_solver --nx 128 --steps 100)
if(STRATA_BUILD_WORKER)
  find_package(Protobuf REQUIRED)
  find_package(gRPC CONFIG QUIET)
  if(NOT gRPC_FOUND)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GRPC REQUIRED IMPORTED_TARGET grpc++)
    add_library(gRPC::grpc++ ALIAS PkgConfig::GRPC)
    find_program(GRPC_CPP_PLUGIN grpc_cpp_plugin REQUIRED)
    add_executable(gRPC::grpc_cpp_plugin IMPORTED)
    set_property(TARGET gRPC::grpc_cpp_plugin PROPERTY IMPORTED_LOCATION "${GRPC_CPP_PLUGIN}")
  endif()
  find_package(CURL REQUIRED)
  find_package(LibArchive REQUIRED)
  find_package(nlohmann_json CONFIG REQUIRED)
  set(PROTO "${CMAKE_CURRENT_SOURCE_DIR}/../proto/engine.proto")
  add_custom_command(OUTPUT engine.pb.cc engine.pb.h engine.grpc.pb.cc engine.grpc.pb.h
    COMMAND protobuf::protoc --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/../proto
      --cpp_out=${CMAKE_CURRENT_BINARY_DIR} --grpc_out=${CMAKE_CURRENT_BINARY_DIR}
      --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> ${PROTO}
    DEPENDS ${PROTO})
  add_library(strata_protocol engine.pb.cc engine.grpc.pb.cc)
  target_include_directories(strata_protocol PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(strata_protocol PUBLIC gRPC::grpc++ protobuf::libprotobuf)
  add_executable(strata-worker-cpp src/main.cpp src/docker.cpp)
  target_include_directories(strata-worker-cpp PRIVATE include)
  target_link_libraries(strata-worker-cpp PRIVATE strata_protocol CURL::libcurl
    LibArchive::LibArchive nlohmann_json::nlohmann_json)
  target_compile_options(strata-worker-cpp PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
